# 计算逻辑
*x进制下的数字*--->*十进制的数值*
每一位上的数字*（位数-1）之和
# 代码实现
输入：十进制的数字N，进制x
输出：N在x进制的表示
~~~C
#include<stdio.h>

char R[36]={"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"};
int N,x;
void f(int p)  
{
	if(p/x==0)
	{
		printf("%c",R[p]);
		return;
	}
	f(p/x);
	printf("%c",R[p%x]);
}

int main()
{
	scanf("%d %d",&N ,&x);
	f(N);
	return 0;
}


~~~
